CREATE TABLE plats (

    nom VARCHAR(200) NOT NULL,

    CONSTRAINT fk_plats_categorie FOREIGN KEY (id_categorie) REFERENCES categories(id_categorie)

-- ============================================================
-- ETAPE 2 : LES CATEGORIES
-- ============================================================

SELECT c.nom_categorie, COUNT(p.id) AS nombre_de_plats
FROM categories c
LEFT JOIN plats p ON c.id_categorie = p.id_categorie
GROUP BY c.id_categorie, c.nom_categorie;


-- ============================================================
-- ETAPE 3 : LES COMMANDES
-- ============================================================

CREATE TABLE commandes (
    id_commande SERIAL PRIMARY KEY,
    date_commande TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE lignes_commande (
    id_commande INT NOT NULL,
    id_plat INT NOT NULL,
    quantite INT NOT NULL,
    PRIMARY KEY (id_commande, id_plat), 
    CONSTRAINT fk_lignes_commandes FOREIGN KEY (id_commande) REFERENCES commandes(id_commande) ON DELETE CASCADE,
    CONSTRAINT fk_lignes_plats FOREIGN KEY (id_plat) REFERENCES plats(id_plat)
);

-- Verification Etape 3 : Montant total de chaque commande
SELECT c.id_commande, c.date_commande, SUM(p.prix * lc.quantite) AS montant_total
FROM commandes c
JOIN lignes_commande lc ON c.id_commande = lc.id_commande
JOIN plats p ON lc.id_plat = p.id_plat
GROUP BY c.id_commande, c.date_commande;

-- Verification Etape 5 : Nombre de commandes et chiffre d'affaires par serveur
SELECT s.nom AS serveur,
       COUNT(DISTINCT c.id_commande) AS nombre_commandes,
       COALESCE(SUM(p.prix * lc.quantite), 0) AS chiffre_affaires
FROM serveurs s
LEFT JOIN commandes c ON s.id_serveur = c.id_serveur
LEFT JOIN lignes_commande lc ON c.id_commande = lc.id_commande
LEFT JOIN plats p ON lc.id_plat = p.id_plat
GROUP BY s.id_serveur, s.nom;
